# The engine plugin loader and runner. core/plugin/engine_plugin.h is what is
# under test; the server's half is tests/rpc/test_rpc_engine_plugins.cpp.
#
# One fixture source built once per defect, on tests/decode/CMakeLists.txt's
# arrangement for the vocoder fixtures and for the reasons it gives, the
# missing platform guard included.
set(revenant_engine_plugin_fixtures
    rv_test_engine_plugin_good
    rv_test_engine_plugin_nosym
    rv_test_engine_plugin_oldabi
    rv_test_engine_plugin_baddescribe
    rv_test_engine_plugin_refuse
    rv_test_engine_plugin_slow
)

foreach(fixture IN LISTS revenant_engine_plugin_fixtures)
    add_library(${fixture} SHARED engine_test_plugin.cpp)
    target_include_directories(${fixture} PRIVATE "${CMAKE_SOURCE_DIR}")
    revenant_apply_warnings(${fixture})
endforeach()

target_compile_definitions(rv_test_engine_plugin_nosym PRIVATE RV_TEST_PLUGIN_OMIT_ON_EVENT)
target_compile_definitions(rv_test_engine_plugin_oldabi PRIVATE RV_TEST_PLUGIN_ABI_VERSION=0u)
target_compile_definitions(rv_test_engine_plugin_baddescribe PRIVATE RV_TEST_PLUGIN_DESCRIBE_FAILS)
target_compile_definitions(rv_test_engine_plugin_refuse PRIVATE RV_TEST_PLUGIN_REFUSE_CREATE)
target_compile_definitions(rv_test_engine_plugin_slow PRIVATE RV_TEST_PLUGIN_SLOW)

# Raw string literals in a generated header, per configuration, for the
# reason tests/decode/CMakeLists.txt gives at length.
file(GENERATE
    OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/engine_test_plugin_paths.h"
    CONTENT [[// Generated by tests/plugin/CMakeLists.txt. Do not edit.
#pragma once

inline constexpr const char* kTestEnginePluginGood = R"($<TARGET_FILE:rv_test_engine_plugin_good>)";
inline constexpr const char* kTestEnginePluginNoSymbol = R"($<TARGET_FILE:rv_test_engine_plugin_nosym>)";
inline constexpr const char* kTestEnginePluginOldAbi = R"($<TARGET_FILE:rv_test_engine_plugin_oldabi>)";
inline constexpr const char* kTestEnginePluginBadDescribe = R"($<TARGET_FILE:rv_test_engine_plugin_baddescribe>)";
inline constexpr const char* kTestEnginePluginRefuse = R"($<TARGET_FILE:rv_test_engine_plugin_refuse>)";
inline constexpr const char* kTestEnginePluginSlow = R"($<TARGET_FILE:rv_test_engine_plugin_slow>)";
]]
)

add_executable(revenant_plugin_tests
    ../support/no_modal_dialogs.cpp
    test_engine_plugin.cpp
)
target_link_libraries(revenant_plugin_tests PRIVATE
    revenant_core
    Catch2::Catch2WithMain
)
target_include_directories(revenant_plugin_tests PRIVATE
    "${CMAKE_SOURCE_DIR}"
    "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>"
)

# Loaded at run time and never linked, so nothing else would build them first.
add_dependencies(revenant_plugin_tests ${revenant_engine_plugin_fixtures})

revenant_apply_warnings(revenant_plugin_tests)

catch_discover_tests(revenant_plugin_tests
    PROPERTIES
        LABELS "plugin"
        TIMEOUT 120
)

# plugins/p25trunk, driven through a host table of the test's own. Loaded at
# run time like the fixtures above, from wherever plugins/CMakeLists.txt put it.
if(TARGET p25trunk)
    file(GENERATE
        OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/p25trunk_path.h"
        CONTENT [[// Generated by tests/plugin/CMakeLists.txt. Do not edit.
#pragma once

inline constexpr const char* kP25TrunkPlugin = R"($<TARGET_FILE:p25trunk>)";
]]
    )
    target_sources(revenant_plugin_tests PRIVATE test_p25trunk.cpp)
    add_dependencies(revenant_plugin_tests p25trunk)
endif()
